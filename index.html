// NOVA FUNÇÃO QUE PUXA HORÁRIO CERTO function aplicarHorarioDaCasa(nomeCasa){ var casa = casas.find(function(c){ return c.nome == nomeCasa }); if(!casa){ // Se não achou com nome exato, tenta achar por P1, P2 casa = casas.find(function(c){ return nomeCasa.includes(c.nome) || c.nome.includes(nomeCasa) }); } if(casa){ document.getElementById('chkAlmoco').checked = !!casa.temAlmoco; document.getElementById('chkJanta').checked = !!casa.temJanta; document.getElementById('eIniA').value = casa.iniA || '12:30'; document.getElementById('eFimA').value = casa.fimA || '15:30'; document.getElementById('eIniJ').value = casa.iniJ || '19:00'; document.getElementById('eFimJ').value = casa.fimJ || '22:00'; calc(); console.log('Horário puxado de '+casa.nome+': '+(casa.temJanta?casa.iniJ+'-'+casa.fimJ:'')+' '+(casa.temAlmoco?casa.iniA+'-'+casa.fimA:'')); return true; } return false; } function setProjeto(p){ projetoAtual=p; document.querySelectorAll('.projBtn').forEach(function(b){b.classList.remove('active')}); var map={'P1':'btnP1','P2':'btnP2','P3':'btnP3','P4':'btnP4','EDILL':'btnEDILL','TAKI SOM':'btnTAKI'}; if(map[p]) document.getElementById(map[p]).classList.add('active'); // SINCRONIZA O SELECT DE BAIXO var selectCasa = document.getElementById('eCasa'); if(selectCasa) selectCasa.value = p; // PUXA HORÁRIO AUTOMÁTICO aplicarHorarioDaCasa(p); } // QUANDO MUDA O SELECT DE BAIXO, TAMBÉM PUXA HORÁRIO document.addEventListener('DOMContentLoaded', function(){ var sel = document.getElementById('eCasa'); if(sel){ sel.addEventListener('change', function(){ var nome = this.value; if(nome){ projetoAtual = nome; // Ativa botão correspondente se tiver document.querySelectorAll('.projBtn').forEach(function(b){b.classList.remove('active')}); var map={'P1':'btnP1','P2':'btnP2','P3':'btnP3','P4':'btnP4','EDILL':'btnEDILL','TAKI SOM':'btnTAKI'}; if(map[nome] && document.getElementById(map[nome])) document.getElementById(map[nome]).classList.add('active'); aplicarHorarioDaCasa(nome); } }); } });
